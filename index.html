// ========================================== // 1. ORIGINAL SHEET MACROS (FOR BUTTONS) // ========================================== function addFoodTruckDates() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const inputSheet = ss.getSheetByName("Test Sheet"); const dataSheet = ss.getSheetByName("Test Data Sheet"); const foodTruck = inputSheet.getRange("B3").getValue(); if (!foodTruck) { SpreadsheetApp.getUi().alert("Please select a Food Truck first."); return; } const inputData = inputSheet.getRange("E5:F").getValues(); const rowsToAdd = []; inputData.forEach(row => { const date = row[0]; const property = row[1]; if (date && property) { rowsToAdd.push([ foodTruck, date, property ]); } }); if (rowsToAdd.length === 0) { SpreadsheetApp.getUi().alert("Please enter at least one Date and Property."); return; } dataSheet .getRange(dataSheet.getLastRow() + 1, 1, rowsToAdd.length, 3) .setValues(rowsToAdd); SpreadsheetApp.getUi().alert(rowsToAdd.length + " visit(s) added successfully."); } function removeFoodTruckDates() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const inputSheet = ss.getSheetByName("Test Sheet"); const dataSheet = ss.getSheetByName("Test Data Sheet"); const foodTruck = inputSheet.getRange("B3").getValue(); if (!foodTruck) { SpreadsheetApp.getUi().alert("Please select a Food Truck first."); return; } const inputData = inputSheet.getRange("E5:F").getValues(); const dataRange = dataSheet.getDataRange(); const data = dataRange.getValues(); let removedCount = 0; for (let i = data.length - 1; i >= 1; i--) { const dataFoodTruck = data[i][0]; const dataDate = data[i][1]; const dataProperty = data[i][2]; for (let j = 0; j < inputData.length; j++) { const inputDate = inputData[j][0]; const inputProperty = inputData[j][1]; if ( inputDate && inputProperty && dataFoodTruck === foodTruck && datesMatch(dataDate, inputDate) && dataProperty === inputProperty ) { dataSheet.deleteRow(i + 1); removedCount++; break; } } } if (removedCount === 0) { SpreadsheetApp.getUi().alert("No matching visits were found."); } else { SpreadsheetApp.getUi().alert(removedCount + " visit(s) removed successfully."); } } function datesMatch(date1, date2) { if (!(date1 instanceof Date) || !(date2 instanceof Date)) { return String(date1) === String(date2); } return ( date1.getFullYear() === date2.getFullYear() && date1.getMonth() === date2.getMonth() && date1.getDate() === date2.getDate() ); } // ========================================== // 2. WEB APP ENDPOINT FOR WEBSITE FORMS // ========================================== function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); // BRANCH 1: "Join Schedule List" if (data.action === "Join Notification List") { var joinSheet = ss.getSheetByName("Notification List"); if (!joinSheet) { joinSheet = ss.insertSheet("Notification List"); joinSheet.appendRow(["Timestamp", "Full Name", "Email Address", "Company", "Property"]); joinSheet.getRange("A1:E1").setFontWeight("bold").setBackground("#f3f4f6"); } joinSheet.appendRow([ new Date(), data.userName, data.userEmail, data.company, data.property ]); } // BRANCH 2: "Recommend a Food Truck" else { var recSheet = ss.getSheetByName("Recommendations"); if (!recSheet) { recSheet = ss.insertSheet("Recommendations"); recSheet.appendRow(["Timestamp", "User Name", "User Email", "Property", "Food Truck Name", "Truck Info / Link"]); recSheet.getRange("A1:F1").setFontWeight("bold").setBackground("#f3f4f6"); } recSheet.appendRow([ new Date(), data.userName, data.userEmail, data.property, data.truckName, data.truckLink ]); } return ContentService .createTextOutput(JSON.stringify({ result: "success" })) .mimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ result: "error", error: error.toString() })) .mimeType(ContentService.MimeType.JSON); } }
